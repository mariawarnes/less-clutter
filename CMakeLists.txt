cmake_minimum_required(VERSION 3.25)
project(LessClutter VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(COMMONLIB_SOURCE "" CACHE PATH "CommonLibSSE-NG / CommonLibVR source")
add_subdirectory("${COMMONLIB_SOURCE}" commonlib)
find_package(spdlog CONFIG REQUIRED)
add_library(LessClutter SHARED src/Plugin.cpp)
target_link_libraries(LessClutter PRIVATE CommonLibSSE::CommonLibSSE spdlog::spdlog)
target_compile_options(LessClutter PRIVATE /Zc:preprocessor /bigobj /EHsc)
install(TARGETS LessClutter RUNTIME DESTINATION SKSE/Plugins)
install(FILES LessClutter.ini DESTINATION SKSE/Plugins)
